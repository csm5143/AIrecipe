document.addEventListener('DOMContentLoaded', () => {
    const fab = document.getElementById('fab-publish');
    const fabIcon = document.getElementById('fab-icon');
    const overlay = document.getElementById('publish-overlay');
    const sheet = document.getElementById('publish-sheet');
    const sheetItems = document.querySelectorAll('.sheet-item');
    let isSheetOpen = false;

    function toggleSheet() {
        isSheetOpen = !isSheetOpen;
        
        if (isSheetOpen) {
            overlay.classList.remove('hidden');
            // small delay to allow display:block to apply before animating opacity
            setTimeout(() => {
                overlay.classList.remove('opacity-0');
                overlay.classList.add('opacity-100');
                overlay.classList.remove('pointer-events-none');
            }, 10);
            
            sheet.classList.remove('translate-y-full');
            fabIcon.style.transform = 'rotate(45deg)';
            
            sheetItems.forEach(item => {
                item.classList.remove('translate-y-8', 'opacity-0');
            });
        } else {
            overlay.classList.remove('opacity-100');
            overlay.classList.add('opacity-0');
            overlay.classList.add('pointer-events-none');
            setTimeout(() => {
                overlay.classList.add('hidden');
            }, 300);
            
            sheet.classList.add('translate-y-full');
            fabIcon.style.transform = 'rotate(0deg)';
            
            sheetItems.forEach(item => {
                item.classList.add('translate-y-8', 'opacity-0');
            });
        }
    }

    fab.addEventListener('click', toggleSheet);
    overlay.addEventListener('click', toggleSheet);
});
</script>
